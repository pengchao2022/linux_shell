allen@ubuntu:~/shell$ echo -n "hello kitty"
hello kittyallen@ubuntu:~/shell$

hello kittyallen@ubuntu:~/shell$ echo -e "Hello \nkitty"
Hello 
kitty

allen@ubuntu:~/shell$ echo -e "\033[44;37m hello,kitty\033[0m"
